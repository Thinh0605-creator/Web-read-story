// SỬA LẠI ĐOẠN ĐỌC DANH SÁCH CHƯƠNG TRONG DETAIL.HTML database.ref(`stories/${storyId}/chapters`).on('value', (snapshot) => { const listContainer = document.getElementById('chapters-list'); listContainer.innerHTML = ""; const chapters = snapshot.val(); if(!chapters) { listContainer.innerHTML = `

Tác giả đang chuẩn bị bản thảo cho chương đầu tiên.

`; return; } // Sắp xếp chương theo số thứ tự chính xác const sortedKeys = Object.keys(chapters).sort((a,b) => parseInt(chapters[a].id) - parseInt(chapters[b].id)); sortedKeys.forEach((chKey) => { const ch = chapters[chKey]; // Tự động ghép nối "Chương + Số chương + : + Tên chương" ở đây const row = ` Chương ${ch.id}: ${ch.title} ${ch.date || ''} `; listContainer.innerHTML += row; }); });
